## BACKEND — TRAITS (App\Traits\)

### HasActivityLog.php
```php
<?php
namespace App\Traits;

use Spatie\Activitylog\Traits\LogsActivity;
use Spatie\Activitylog\LogOptions;
use App\Services\ThreatDetectionService;

trait HasActivityLog
{
    use LogsActivity;

    public function getActivitylogOptions(): LogOptions
    {
        return LogOptions::defaults()
            ->logAll()
            ->logOnlyDirty()
            ->setDescriptionForEvent(fn(string $eventName) => 
                "{$eventName} " . class_basename($this));
    }

    /**
     * Log a custom security activity for this user/model.
     * Also dispatches threat detection after logging.
     */
    public function logSecurityActivity(
        string $action,
        string $description,
        string $module = 'system',
        int $riskScore = 0
    ): void {
        $log = \App\Models\ActivityLog::create([
            'user_id'                => auth()->id(),
            'action'                 => $action,
            'module'                 => $module,
            'description'            => $description,
            'ip_address'             => request()->ip(),
            'user_agent'             => request()->userAgent(),
            'risk_score_contribution'=> $riskScore,
            'alert_triggered'        => false,
        ]);

        if ($log && auth()->check()) {
            \App\Jobs\AnalyzeThreatJob::dispatch($log, auth()->user());
        }
    }
}
```

### HasRiskScore.php
```php
<?php
namespace App\Traits;

trait HasRiskScore
{
    /**
     * Calculate total risk score from activity_logs in last 24 hours.
     */
    public function calculateRiskScore(): int { ... }

    /**
     * Returns true if current_score >= system config threshold.
     */
    public function isHighRisk(): bool { ... }

    /**
     * Kill switch: lock account, terminate sessions, broadcast, notify.
     */
    public function triggerKillSwitch(string $reason): void
    {
        $this->update([
            'is_locked'   => true,
            'locked_at'   => now(),
            'lock_reason' => $reason,
        ]);

        // Delete all Laravel sessions for this user
        \DB::table('sessions')
            ->where('user_id', $this->id)
            ->delete();

        // Mark all tracker sessions as force terminated
        $this->sessionTracker()
            ->where('is_active', true)
            ->update([
                'is_active'             => false,
                'logout_at'             => now(),
                'was_force_terminated'  => true,
            ]);

        // Create threat alert record
        \App\Models\ThreatAlert::create([
            'user_id'          => $this->id,
            'alert_type'       => 'privilege_escalation',
            'severity'         => 'critical',
            'status'           => 'open',
            'auto_mitigated'   => true,
            'mitigation_action'=> 'account_locked_kill_switch',
            'notes'            => $reason,
        ]);

        // Broadcast to admin WebSocket channel
        broadcast(new \App\Events\AccountLockedEvent($this, $reason));

        // Send notifications
        $admins = \App\Models\User::role(['admin', 'superadmin'])->get();
        \Illuminate\Support\Facades\Notification::send(
            $admins,
            new \App\Notifications\AccountLockedNotification($this, $reason)
        );
    }

    public function riskScore()
    {
        return $this->hasOne(\App\Models\UserRiskScore::class);
    }

    public function sessionTracker()
    {
        return $this->hasMany(\App\Models\UserSessionsTracker::class);
    }
}
```

## BACKEND — FORTIFY INTEGRATION

In app/Providers/FortifyServiceProvider.php, ADD (do not replace):

```php
// Hook login: check lock, register session, increment fail counter
Fortify::authenticateUsing(function (Request $request) {
    $user = User::where('email', $request->email)->first();

    if (!$user) return null;

    // Check lock status before even verifying password
    if ($user->is_locked) {
        throw \Illuminate\Validation\ValidationException::withMessages([
            'email' => ['Your account has been locked. Reason: ' . $user->lock_reason
                        . ' Please contact the ICT department.'],
        ]);
    }

    if (Hash::check($request->password, $user->password)) {
        // Successful login
        $user->update([
            'failed_login_count' => 0,
            'last_login_at'      => now(),
            'last_login_ip'      => $request->ip(),
        ]);
        app(SessionTrackerService::class)->registerLogin($user, $request);
        app(SessionTrackerService::class)->detectSimultaneousSessions($user, $request->ip());
        return $user;
    }

    // Failed login — increment counter and check threshold
    $user->increment('failed_login_count');
    $user->logSecurityActivity('failed_login', 'Failed login attempt', 'auth', 15);

    $maxFails = (int) SystemConfiguration::getValue('max_failed_logins', 5);
    if ($user->failed_login_count >= $maxFails) {
        $user->triggerKillSwitch("Auto-locked after {$user->failed_login_count} failed logins");
    }

    return null;
});

// Hook logout: register logout in session tracker
Fortify::logoutResponse(function (Request $request) {
    if (auth()->check()) {
        app(SessionTrackerService::class)->registerLogout(auth()->user());
    }
    return redirect('/login');
});
```

        });
